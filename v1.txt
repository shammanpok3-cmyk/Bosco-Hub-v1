-- UTILITY
local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Blacklist
rayParams.IgnoreWater = true
local function isTargetVisible(origin, targetPos, targetPlayerObj)
    local ignoreList = {LP.Character}
    if targetPlayerObj and targetPlayerObj.Character then table.insert(ignoreList, targetPlayerObj.Character) end
    rayParams.FilterDescendantsInstances = ignoreList
    local dir = (targetPos - origin)
    if dir.Magnitude <= 0 then return true end
    local result = Workspace:Raycast(origin, dir.Unit * dir.Magnitude, rayParams)
    if not result then return true end
    local hit = result.Instance
    local isSmoke = hit.Transparency > 0.5 or not hit.CanCollide or hit.Name:lower():find("smoke") or hit.Name:lower():find("effect") or hit.Name:lower():find("particle")
    if not isSmoke and hit.Parent then isSmoke = hit.Parent.Name:lower():find("smoke") or hit.Parent.Name:lower():find("grenade") end
    if isSmoke then
        local newIgnoreList = {}
        for _, v in ipairs(ignoreList) do table.insert(newIgnoreList, v) end
        table.insert(newIgnoreList, hit)
        rayParams.FilterDescendantsInstances = newIgnoreList
        local newOrigin = result.Position + dir.Unit * 0.1
        local newDist = (targetPos - newOrigin).Magnitude
        if newDist <= 0 then return true end
        return Workspace:Raycast(newOrigin, dir.Unit * newDist, rayParams) == nil
    end
    return false
end

Features.infiniteJumpEnabled = false

-- State for storing original item info
local OriginalItemInfo = setmetatable({}, { __mode = 'k' })
local InfoModifierFighter = nil
local InfoModifierItemAdded = nil
local InfoModifierItemRemoved = nil

-- Helper to resolve local fighter
local function ResolveLocalFighter()
    local char = LP.Character
    if not char then
        return nil
    end
    
    local fighter = char:FindFirstChild("Fighter")
    if not fighter then
        fighter = char:FindFirstChild("FighterController")
    end
    if not fighter then
        local replicatedStorage = game:GetService("ReplicatedStorage")
        local modules = replicatedStorage:FindFirstChild("Modules")
        if modules then
            local utility = modules:FindFirstChild("Utility")
            if utility then
                pcall(function()
                    local required = require(utility)
                    if required and required.LocalFighterController then
                        fighter = required.LocalFighterController
                    end
                end)
            end
        end
    end
    
    return fighter
end

-- Store original item info before modification
local function StoreOriginalItemInfo(item)
    if type(item) ~= 'table' or OriginalItemInfo[item] ~= nil then
        return
    end
    
    local info = item.Info
    if type(info) ~= 'table' then
        return
    end
    
    OriginalItemInfo[item] = {
        Info = info,
        HasMaxDoubleJumps = rawget(info, 'MaxDoubleJumps') ~= nil,
        MaxDoubleJumps = rawget(info, 'MaxDoubleJumps'),
    }
end

-- Apply Infinite Double Jumps modifier to an item
local function ApplyInfiniteDoubleJumps(item)
    StoreOriginalItemInfo(item)
    
    local original = OriginalItemInfo[item]
    local info = original and original.Info or nil
    if type(info) ~= 'table' then
        return
    end
    
    if Features.infiniteJumpEnabled then
        rawset(info, 'MaxDoubleJumps', 676767)
    end
end

-- Restore original item info
local function RestoreOriginalItemInfo(item)
    local original = OriginalItemInfo[item]
    local info = original and original.Info or nil
    
    if type(info) == 'table' then
        if original.HasMaxDoubleJumps then
            rawset(info, 'MaxDoubleJumps', original.MaxDoubleJumps)
        else
            rawset(info, 'MaxDoubleJumps', nil)
        end
    end
    
    OriginalItemInfo[item] = nil
end

-- Restore all modified items
local function RestoreAllItems()
    for item, _ in pairs(OriginalItemInfo) do
        RestoreOriginalItemInfo(item)
    end
    
    if InfoModifierItemAdded then
        InfoModifierItemAdded:Disconnect()
        InfoModifierItemAdded = nil
    end
    
    if InfoModifierItemRemoved then
        InfoModifierItemRemoved:Disconnect()
        InfoModifierItemRemoved = nil
    end
    
    InfoModifierFighter = nil
end

-- Refresh modifiers for all items
local function RefreshInfiniteDoubleJumps()
    local fighter = ResolveLocalFighter()
    
    if fighter ~= InfoModifierFighter then
        RestoreAllItems()
        InfoModifierFighter = fighter
        
        if fighter then
            if fighter.ItemAdded and type(fighter.ItemAdded.Connect) == 'function' then
                InfoModifierItemAdded = fighter.ItemAdded:Connect(function(item)
                    if Features.infiniteJumpEnabled then
                        ApplyInfiniteDoubleJumps(item)
                    end
                end)
            end
            
            if fighter.ItemRemoved and type(fighter.ItemRemoved.Connect) == 'function' then
                InfoModifierItemRemoved = fighter.ItemRemoved:Connect(function(item)
                    RestoreOriginalItemInfo(item)
                end)
            end
        end
    end
    
    if not fighter or type(fighter.Items) ~= 'table' then
        return false
    end
    
    for _, item in pairs(fighter.Items) do
        if Features.infiniteJumpEnabled then
            ApplyInfiniteDoubleJumps(item)
        else
            RestoreOriginalItemInfo(item)
        end
    end
    
    return true
end

-- Set Infinite Double Jumps
function Features.setInfiniteJump(on)
    Features.infiniteJumpEnabled = on
    
    if on then
        RefreshInfiniteDoubleJumps()
        task.delay(0.5, function()
            if Features.infiniteJumpEnabled then
                RefreshInfiniteDoubleJumps()
            end
        end)
    else
        RestoreAllItems()
    end
end

-- Handle character changes
LP.CharacterAdded:Connect(function(character)
    task.wait(0.5)
    
    if Features.infiniteJumpEnabled then
        RefreshInfiniteDoubleJumps()
        task.delay(1, function()
            if Features.infiniteJumpEnabled then
                RefreshInfiniteDoubleJumps()
            end
        end)
    else
        RestoreAllItems()
    end
end)

-- Watch for new items
RunService.RenderStepped:Connect(function()
    if not Features.infiniteJumpEnabled then
        return
    end
    
    if tick() % 2 < 0.03 then
        RefreshInfiniteDoubleJumps()
    end
end)

local function GetVisibilityCache(character)
    local cache = AimbotVisibilityCache[character]
    if not cache then
        cache = { visible = false, visibleAt = 0, hiddenAt = 0 }
        AimbotVisibilityCache[character] = cache
    end
    return cache
end

local function IsPointVisibleCached(character, worldPoint)
    local cache = GetVisibilityCache(character)
    local now = tick()
    
    if cache.lastCheck and (now - cache.lastCheck) < AIMBOT_VISIBLE_CACHE_WINDOW then
        return cache.visible
    end
    
    cache.lastCheck = now
    cache.visible = isTargetVisible(Camera.CFrame.Position, worldPoint, 
        Players:GetPlayerFromCharacter(character))
    
    return cache.visible
end

local function ResolveAimbotTargetPart(subject, aimPart, mousePosition, maxDistance, fovRadius, ignoreFov, camera)
    if not subject or not subject.Parent then
        return nil, nil, false
    end
    
    local humanoid = subject:FindFirstChildOfClass("Humanoid")
    if not humanoid or humanoid.Health <= 0 then
        return nil, nil, false
    end
    
    if subject:FindFirstChildOfClass("ForceField") then
        return nil, nil, false
    end
    
    if Features.silentAimSettings.CheckRiotShield then
        local riotShield = subject:FindFirstChild("RiotShield") or 
                          subject:FindFirstChild("Shield") or
                          subject:FindFirstChild("Riot Shield")
        if riotShield and riotShield:IsA("BasePart") then
            local shieldPos = riotShield.Position
            local targetPos = subject:FindFirstChild("Head") and subject.Head.Position or subject:GetPivot().Position
            local cameraPos = camera.CFrame.Position
            
            local toTarget = (targetPos - cameraPos).Unit
            local toShield = (shieldPos - cameraPos).Unit
            if toTarget:Dot(toShield) > 0.9 then
                return nil, nil, false
            end
        end
    end
    
    if Features.silentAimSettings.CheckKatanaDeflect then
        local katana = subject:FindFirstChild("Katana") or 
                      subject:FindFirstChildOfClass("Tool")
        if katana and katana:FindFirstChild("Deflecting") then
            return nil, nil, false
        end
    end
    
    local selectedPart = nil
    local worldPosition = nil
    
    if aimPart == "Auto" then
        selectedPart = subject:FindFirstChild("Head") or 
                      subject:FindFirstChild("HumanoidRootPart") or
                      subject:FindFirstChild("UpperTorso")
    elseif aimPart == "Random" then
        local randomState = AimbotSilentState.RandomTarget
        local randomPart = AimbotSilentState.RandomPart
        
        if randomState == subject and randomPart and randomPart.Parent then
            selectedPart = randomPart
        else
            local parts = {}
            for _, name in ipairs({"Head", "HumanoidRootPart", "UpperTorso", "LowerTorso", "LeftArm", "RightArm", "LeftLeg", "RightLeg"}) do
                local part = subject:FindFirstChild(name)
                if part then
                    table.insert(parts, part)
                end
            end
            if #parts > 0 then
                selectedPart = parts[math.random(1, #parts)]
                AimbotSilentState.RandomTarget = subject
                AimbotSilentState.RandomPart = selectedPart
            end
        end
    elseif aimPart == "Closest" then
        local closestDist = math.huge
        for _, part in ipairs(subject:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then
                local dist = (camera.CFrame.Position - part.Position).Magnitude
                if dist < closestDist then
                    closestDist = dist
                    selectedPart = part
                end
            end
        end
    else
        selectedPart = subject:FindFirstChild(aimPart) or
                      subject:FindFirstChild(aimPart, true)
    end
    
    if not selectedPart or not selectedPart:IsA("BasePart") then
        return nil, nil, false
    end
    
    worldPosition = selectedPart.Position
    
    if Features.silentAimSettings.Prediction and selectedPart:IsA("BasePart") then
        local velocity = selectedPart.AssemblyLinearVelocity
        if velocity and velocity.Magnitude > 0.5 then
            local distance = (camera.CFrame.Position - worldPosition).Magnitude
            local bulletSpeed = 450
            
            local char = LP.Character
            if char then
                local tool = char:FindFirstChildOfClass("Tool")
                if tool then
                    local config = tool:FindFirstChild("Configuration") or tool:FindFirstChild("Stats")
                    if config then
                        local bs = config:FindFirstChild("BulletSpeed") or 
                                  config:FindFirstChild("ProjectileSpeed") or 
                                  config:FindFirstChild("Velocity")
                        if bs and bs.Value then
                            bulletSpeed = bs.Value
                        end
                    end
                end
            end
            
            local timeToTarget = distance / bulletSpeed
            worldPosition = worldPosition + (velocity * timeToTarget * 0.4)
        end
    end
    
    local isVisible = true
    if Features.silentAimSettings.WallCheck then
        isVisible = IsPointVisibleCached(subject, worldPosition)
    end
    
    return selectedPart, worldPosition, isVisible
end

local function BuildAimbotTargetInfo(subject, aimPart, mousePosition, maxDistance, fovRadius, ignoreFov, player, camera)
    local part, worldPosition, isVisible = ResolveAimbotTargetPart(
        subject, aimPart, mousePosition, maxDistance, fovRadius, ignoreFov, camera
    )
    
    if not part or not worldPosition then
        return nil
    end
    
    if not isVisible then
        return nil
    end
    
    local rootPosition = subject:FindFirstChild("HumanoidRootPart")
    if rootPosition then
        local worldDistance = (worldPosition - rootPosition.Position).Magnitude
        if worldDistance > maxDistance then
            return nil
        end
    end
    
    local screenPosition, onScreen = camera:WorldToViewportPoint(worldPosition)
    local isOnScreen = onScreen and screenPosition.Z > 0
    
    if not ignoreFov and not isOnScreen then
        return nil
    end
    
    local screenVector = Vector2.new(screenPosition.X, screenPosition.Y)
    local screenDistance = (screenVector - mousePosition).Magnitude
    
    if not ignoreFov and screenDistance > fovRadius then
        return nil
    end
    
    local worldDistance = (camera.CFrame.Position - worldPosition).Magnitude
    
    return {
        instance = subject,
        part = part,
        player = player,
        worldPosition = worldPosition,
        worldDistance = worldDistance,
        screenDistance = screenDistance,
        selectionDistance = isOnScreen and screenDistance or (AIMBOT_OFFSCREEN_SELECTION_PENALTY + worldDistance),
        displayName = player and (player.DisplayName or player.Name) or subject.Name,
        isVisible = isVisible,
    }
end

local function ConsiderBestAimbotTarget(bestTarget, subject, aimPart, mousePosition, maxDistance, fovRadius, ignoreFov, player, camera)
    local targetInfo = BuildAimbotTargetInfo(
        subject, aimPart, mousePosition, maxDistance, fovRadius, ignoreFov, player, camera
    )
    
    if targetInfo and (not bestTarget or targetInfo.selectionDistance < bestTarget.selectionDistance) then
        return targetInfo
    end
    
    return bestTarget
end

local function GetBestSilentAimTarget(requireVisible)
    local aimPart = Features.silentAimSettings.HitPart or "Auto"
    local fovRadius = Features.silentAimSettings.FOV
    local maxDistance = Features.silentAimSettings.MaxDistance
    local ignoreFov = Features.silentAimSettings.IgnoreFov or false
    local mousePosition = UIS:GetMouseLocation()
    local camera = Camera
    
    if not camera then
        return nil
    end
    
    local bestTarget = nil
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            local character = player.Character
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            
            if humanoid and humanoid.Health > 0 then
                if Features.silentAimSettings.TeamCheck and isTeammate(player) then
                    continue
                end
                
                bestTarget = ConsiderBestAimbotTarget(
                    bestTarget, character, aimPart, mousePosition, 
                    maxDistance, fovRadius, ignoreFov, player, camera
                )
            end
        end
    end
    
    return bestTarget
end

local function CanUseSilentAimWithItem(item)
    if not item then
        return false
    end
    
    if AIMBOT_HARD_BLOCK_ITEMS[item.Name] == true then
        return false
    end
    
    local ignoreWeapons = Features.silentAimSettings.IgnoreWeapons or {}
    if type(ignoreWeapons) == "table" then
        for _, weaponName in ipairs(ignoreWeapons) do
            if item.Name == weaponName then
                return false
            end
        end
    end
    
    if type(item.StartShooting) == "function" then
        return true
    end
    
    if item.Name == "Daggers" or item.Name == "Bow" or item.Name == "Slingshot" then
        return true
    end
    
    if AIMBOT_HOLD_FIRE_ITEMS[item.Name] == true then
        return true
    end
    
    return false
end

local function BeginAimbotSilentShot(item)
    if (AimbotSilentState.ShotPrimeDepth or 0) > 0 then
        return false, nil, nil
    end
    
    if not Features.silentAimEnabled then
        return false, nil, nil
    end
    
    if not CanUseSilentAimWithItem(item) then
        return false, nil, nil
    end
    
    local targetInfo = GetBestSilentAimTarget(true)
    if not targetInfo then
        return false, nil, nil
    end
    
    if targetInfo.player and targetInfo.player.Character then
        if targetInfo.player.Character:FindFirstChildOfClass("ForceField") then
            return false, nil, nil
        end
    end
    
    local camera = Camera
    local originalCFrame = camera and camera.CFrame or nil
    if not camera or not originalCFrame then
        return false, nil, nil
    end
    
    local targetPos = targetInfo.worldPosition
    camera.CFrame = CFrame.lookAt(camera.CFrame.Position, targetPos)
    
    AimbotSilentState.ShotPrimeDepth = (AimbotSilentState.ShotPrimeDepth or 0) + 1
    
    return true, camera, originalCFrame
end

local function FinishAimbotSilentShot(didPrime, camera, originalCFrame)
    if not didPrime then
        return
    end
    
    AimbotSilentState.ShotPrimeDepth = math.max((AimbotSilentState.ShotPrimeDepth or 1) - 1, 0)
    
    if camera and originalCFrame and Camera == camera then
        camera.CFrame = originalCFrame
    end
end

local function EnsureSilentAimStartShootingHook(item)
    if AimbotSilentState.HookedItem == item then
        return
    end
    
    if not CanUseSilentAimWithItem(item) then
        return
    end
    
    if not item or type(item.StartShooting) ~= "function" then
        return
    end
    
    local originalStartShooting = item.StartShooting
    local wrappedStartShooting
    wrappedStartShooting = function(self, ...)
        local didPrime, camera, originalCFrame = BeginAimbotSilentShot(self)
        local results = {pcall(originalStartShooting, self, ...)}
        FinishAimbotSilentShot(didPrime, camera, originalCFrame)
        if not results[1] then
            return nil
        end
        return table.unpack(results, 2, #results)
    end
    
    AimbotSilentState.HookedItem = item
    AimbotSilentState.OriginalStartShooting = originalStartShooting
    AimbotSilentState.WrappedStartShooting = wrappedStartShooting
    item.StartShooting = wrappedStartShooting
end

local function ReleaseTriggerbotMouseHold()
    if AimbotSilentState.TriggerbotHoldingMouse then
        mouse1release()
        AimbotSilentState.TriggerbotHoldingMouse = false
        AimbotSilentState.TriggerbotHoldX = nil
        AimbotSilentState.TriggerbotHoldY = nil
    end
end

local function FireTriggerbotShot()
    local now = tick()
    if (now - AimbotTriggerbotLastShotAt) < 0.05 then
        return false
    end
    
    mouse1press()
    task.wait(0.01)
    mouse1release()
    
    AimbotTriggerbotLastShotAt = now
    return true
end

local function TryTriggerbotShot(targetInfo, item)
    if not Features.silentAimSettings.TriggerbotEnabled then
        return false
    end
    
    if not targetInfo or not item then
        ReleaseTriggerbotMouseHold()
        return false
    end
    
    local now = tick()
    local targetIdentity = targetInfo.player or targetInfo.instance or targetInfo.part
    local reactionTime = Features.silentAimSettings.TriggerReactionTime or 0
    
    if AimbotSilentState.TriggerbotTarget ~= targetIdentity then
        AimbotSilentState.TriggerbotTarget = targetIdentity
        AimbotSilentState.TriggerbotTargetAcquiredAt = now
        if reactionTime > 0 then
            ReleaseTriggerbotMouseHold()
            return false
        end
    elseif (now - AimbotSilentState.TriggerbotTargetAcquiredAt) < reactionTime then
        ReleaseTriggerbotMouseHold()
        return false
    end
    
    return FireTriggerbotShot()
end

local silentAimConnection = nil
local function UpdateSilentAim()
    if not Features.silentAimEnabled then
        return
    end
    
    local char = LP.Character
    if not char then
        return
    end
    
    local item = char:FindFirstChildOfClass("Tool")
    if item and CanUseSilentAimWithItem(item) then
        EnsureSilentAimStartShootingHook(item)
    end

-- AIMBOT (Camera Aim)
Features.aimbotEnabled = false
Features.aimbotSettings = { FOV=120, AimSpeed=15, TargetPart="Head", ShowFOV=true, MaxDistance=500, WallCheck=true, TeamCheck=true, Prediction=true, StickyAim=true, StickyTime=1.5, TargetPriority="FOV", AutoFire=false, AutoFireDelay=0.100, AutoFireHold=false, RequireRightClick=true, ADSEnabled=true, ADSMultiplier=80, PostKillInertia=true }
local function getAdaptiveSpeed(baseSpeed)
    local sens = UIS.MouseDeltaSensitivity or 0.5
    local multiplier = 0.5 / math.max(sens, 0.01)
    if Features.aimbotSettings.ADSEnabled and Features.aimbotSettings.RequireRightClick and UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
        local adsVal = math.max(Features.aimbotSettings.ADSMultiplier or 100, 1)
        multiplier = multiplier * (100 / adsVal)
    end
    return baseSpeed * multiplier * 5
end
local lockedTarget, targetPlayer, lockTime, onTargetTime, aimbotConnection, fovCircle = nil, nil, 0, nil, nil, nil
local function updateFOVCircle()
    if not hasDrawing then return end
    if not Features.aimbotSettings.ShowFOV then if fovCircle then fovCircle.Visible=false end; return end
    if not fovCircle then fovCircle=Drawing.new("Circle"); fovCircle.Thickness=1.2; fovCircle.NumSides=100; fovCircle.Filled=false; fovCircle.ZIndex=999; fovCircle.Transparency=0.3 end
    fovCircle.Visible=true; fovCircle.Radius=Features.aimbotSettings.FOV
    local mp=UIS:GetMouseLocation(); fovCircle.Position=Vector2.new(mp.X,mp.Y)
    fovCircle.Color=lockedTarget and Color3.fromRGB(100,255,100) or Color3.fromRGB(255,255,255)
end
local function isValidTarget(part) if not part or not part.Parent then return false end; local h=part.Parent:FindFirstChildOfClass("Humanoid"); return h and h.Health>0 and not part.Parent:FindFirstChildOfClass("ForceField") end
local function getPredictedPosition(part)
    local pos = part.Position
    if not Features.aimbotSettings.Prediction then return pos end
    local vel = part.AssemblyLinearVelocity
    if vel.Magnitude < 0.5 then return pos end
    local cam = Camera; if not cam then return pos end
    local distance = (cam.CFrame.Position - pos).Magnitude
    local bulletSpeed = 450
    local char = LP.Character
    if char then local tool = char:FindFirstChildOfClass("Tool")
        if tool then local config = tool:FindFirstChild("Configuration") or tool:FindFirstChild("Stats")
            if config then local bs = config:FindFirstChild("BulletSpeed") or config:FindFirstChild("ProjectileSpeed") or config:FindFirstChild("Velocity")
                if bs and bs.Value then bulletSpeed = bs.Value end
            end
        end
    end
    local timeToTarget = distance / bulletSpeed
    return pos + (vel * timeToTarget * 0.4)
end
local function getBestTarget()
    local cam=Camera; if not cam then return nil,nil end
    local mp=UIS:GetMouseLocation(); local fovCenter=Vector2.new(mp.X,mp.Y); local s=Features.aimbotSettings; local now=tick()
    if s.StickyAim and lockedTarget and isValidTarget(lockedTarget) and now-lockTime<s.StickyTime then
        local tPlayer=Players:GetPlayerFromCharacter(lockedTarget.Parent)
        if not (s.TeamCheck and isTeammate(tPlayer)) then
            local aimPos=getPredictedPosition(lockedTarget)
            if (cam.CFrame.Position-aimPos).Magnitude<=s.MaxDistance then
                local sp,on=cam:WorldToViewportPoint(aimPos)
                if on and sp.Z>0 and (Vector2.new(sp.X,sp.Y)-fovCenter).Magnitude<=s.FOV*1.3 then
                    if not s.WallCheck or isTargetVisible(cam.CFrame.Position,aimPos,tPlayer) then targetPlayer=tPlayer; return lockedTarget,Vector2.new(sp.X,sp.Y) end
                end
            end
        end
        lockedTarget=nil; targetPlayer=nil
    end
    local bestPart, bestDist = nil, s.FOV
    for _,p in ipairs(Players:GetPlayers()) do
        if p==LP then continue end
        if s.TeamCheck and isTeammate(p) then continue end
        local char=p.Character; if not char then continue end
        local hum=char:FindFirstChildOfClass("Humanoid"); if hum and hum.Health<=0 then continue end
        local part=char:FindFirstChild(s.TargetPart) or char:FindFirstChild("Head")
        if not part then continue end
        local aimPos = getPredictedPosition(part)
        if (cam.CFrame.Position-aimPos).Magnitude>s.MaxDistance then continue end
        local sp, onScreen = cam:WorldToViewportPoint(aimPos)
        if not onScreen or sp.Z<=0 then continue end
        local dist2D = (Vector2.new(sp.X,sp.Y)-fovCenter).Magnitude
        if dist2D>s.FOV then continue end
        if s.WallCheck and not isTargetVisible(cam.CFrame.Position,aimPos,p) then continue end
        if dist2D < bestDist then bestDist = dist2D; bestPart = part end
    end
    if not bestPart then return nil,nil end
    lockedTarget=bestPart; lockTime=now
    local aimPos = getPredictedPosition(bestPart)
    local sp = cam:WorldToViewportPoint(aimPos)
    return bestPart, Vector2.new(sp.X, sp.Y)
end
local function moveMouseToTarget(targetScreenPos,overrideSpeed)
    local mp=UIS:GetMouseLocation(); local cm=Vector2.new(mp.X,mp.Y); local delta=targetScreenPos-cm; local dist=delta.Magnitude
    if dist<2 then return true end
    local speed = getAdaptiveSpeed(overrideSpeed or Features.aimbotSettings.AimSpeed)
    local distMult
    if dist < 30 then distMult = 0.65
    elseif dist < 80 then distMult = 0.80
    elseif dist < 200 then distMult = 0.90
    else distMult = 1.0 end
    local moveAmount = math.min(speed * 0.8, dist * 0.95) * distMult
    local dir = delta.Unit; mousemoverel(dir.X * moveAmount, dir.Y * moveAmount)
    return false
end
local function handleAutoFire(targetScreenPos)
    if not Features.aimbotSettings.AutoFire then onTargetTime=nil; return end
    local mp=UIS:GetMouseLocation(); local cm=Vector2.new(mp.X,mp.Y); local dist=(cm-targetScreenPos).Magnitude; local now=tick()
    if dist<20 then
        if not onTargetTime then onTargetTime=now end
        if now-onTargetTime >= Features.aimbotSettings.AutoFireDelay then
            if Features.aimbotSettings.AutoFireHold then mouse1press() else 
                mouse1press(); task.wait(0.01); mouse1release(); onTargetTime=now
                if Features.aimbotSettings.PostKillInertia then
                    local driftX = (math.random() - 0.5) * 10
                    local driftY = math.random(2, 6)
                    mousemoverel(driftX, driftY)
                end
            end
        end
    else onTargetTime=nil; if Features.aimbotSettings.AutoFireHold then mouse1release() end end
end
local function aimbotRenderStep()
    if not Features.aimbotEnabled then if fovCircle then fovCircle.Visible=false end; return end
    local char = LP.Character
    if not char or not char:FindFirstChildOfClass("Humanoid") or char:FindFirstChildOfClass("Humanoid").Health <= 0 then return end
    updateFOVCircle()
    if Features.aimbotSettings.RequireRightClick and not UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
        lockedTarget=nil; targetPlayer=nil; onTargetTime=nil; return
    end
    local targetPart,targetScreenPos=getBestTarget()
    if not targetScreenPos then return end
    moveMouseToTarget(targetScreenPos); handleAutoFire(targetScreenPos)
end
local function startAimbotLoop() if aimbotConnection then return end; aimbotConnection=RunService.RenderStepped:Connect(aimbotRenderStep) end
local function stopAimbotLoop() if aimbotConnection then aimbotConnection:Disconnect(); aimbotConnection=nil end; lockedTarget=nil; targetPlayer=nil; onTargetTime=nil; mouse1release(); if fovCircle then fovCircle.Visible=false end end
function Features.setAimbot(on) Features.aimbotEnabled=on; if on then startAimbotLoop() else stopAimbotLoop() end end

local function doFlick()
    if isFlicking then return end; isFlicking = true
    local targetHead = nil
    local cam = Camera; local s = Features.flickSettings
    if cam then
        local bestDist = s.FOV
        for _, p in ipairs(Players:GetPlayers()) do
            if p == LP then continue end
            if s.TeamCheck and isTeammate(p) then continue end
            local char = p.Character; if not char then continue end
            local head = char:FindFirstChild("Head"); if not head then continue end
            local hum = char:FindFirstChildOfClass("Humanoid"); if hum and hum.Health <= 0 then continue end
            if (cam.CFrame.Position - head.Position).Magnitude > s.MaxDistance then continue end
            if s.WallCheck and not isTargetVisible(cam.CFrame.Position, head.Position, p) then continue end
            local aimPos = head.Position
            if Features.aimbotSettings.Prediction then
                local vel = head.AssemblyLinearVelocity
                if vel.Magnitude > 0.5 then
                    local distance = (cam.CFrame.Position - head.Position).Magnitude; local bulletSpeed = 450
                    local c = LP.Character
                    if c then local t = c:FindFirstChildOfClass("Tool")
                        if t then local cfg = t:FindFirstChild("Configuration") or t:FindFirstChild("Stats")
                            if cfg then local bs = cfg:FindFirstChild("BulletSpeed") or cfg:FindFirstChild("ProjectileSpeed") or cfg:FindFirstChild("Velocity")
                                if bs and bs.Value then bulletSpeed = bs.Value end
                            end
                        end
                    end
                    aimPos = head.Position + (vel * (distance / bulletSpeed) * 0.4)
                end
            end
            local sp, on = cam:WorldToViewportPoint(aimPos)
            if on then
                local d = math.sqrt((sp.X - UIS:GetMouseLocation().X)^2 + (sp.Y - UIS:GetMouseLocation().Y)^2)
                if d < bestDist then bestDist = d; targetHead = head end
            end
        end
    end
    if not targetHead then isFlicking = false; return end
    
    local startX, startY = UIS:GetMouseLocation().X, UIS:GetMouseLocation().Y
    local sp, on = cam:WorldToViewportPoint(targetHead.Position)
    if not on then isFlicking = false; return end
    
    local dx = sp.X - startX
    local dy = sp.Y - startY
    mousemoverel(dx, dy)
    
    local delay = 0.02 / (s.Speed / 25)
    task.wait(delay)
    
    local wasSilent = Features.silentAimEnabled
    Features.silentAimEnabled = true
    mouse1press()
    mousemoverel(-dx, -dy)
    mouse1release()
    Features.silentAimEnabled = wasSilent
    isFlicking = false
end

if isMobile then
    mobileMenuButton = Instance.new("TextButton")
    mobileMenuButton.Name = "MobileMenuButton"
    mobileMenuButton.Size = UDim2.fromOffset(44, 44)
    mobileMenuButton.Position = UDim2.new(0, 20, 0, 100)
    mobileMenuButton.BackgroundColor3 = C.accent
    mobileMenuButton.Text = "☰"
    mobileMenuButton.Font = FONT_BOL
    mobileMenuButton.TextSize = 20
    mobileMenuButton.TextColor3 = Color3.fromRGB(255, 255, 255)
    mobileMenuButton.AutoButtonColor = false
    mobileMenuButton.ZIndex = 1000
    mobileMenuButton.Parent = gui
    
    mkCorner(mobileMenuButton, UDim.new(1, 0))
    mkStroke(mobileMenuButton, C.accentBrt, 0, 2)
    mkGradient(mobileMenuButton, C.accent, C.accent2)
    
    mobileMenuButton.BackgroundTransparency = 0.2
    
    local btnDragging = false
    local btnDragStart = nil
    local btnDragOffset = nil
    local wasDragged = false
    
    mobileMenuButton.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            btnDragging = true
            wasDragged = false
            btnDragStart = input.Position
            mobileMenuButton.BackgroundTransparency = 0
            
            local mousePos = Vector2.new(input.Position.X, input.Position.Y)
            local btnPos = Vector2.new(mobileMenuButton.AbsolutePosition.X, mobileMenuButton.AbsolutePosition.Y)
            btnDragOffset = mousePos - btnPos
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    btnDragging = false
                    mobileMenuButton.BackgroundTransparency = 0.2
                end
            end)
        end
    end)
    
    UIS.InputChanged:Connect(function(input)
        if btnDragging and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
            local newPos = Vector2.new(input.Position.X, input.Position.Y) - btnDragOffset
            
            if btnDragStart and (Vector2.new(input.Position.X, input.Position.Y) - btnDragStart).Magnitude > 5 then
                wasDragged = true
            end
            
            local screenSize = Camera.ViewportSize
            local btnSize = Vector2.new(44, 44)
            newPos = Vector2.new(
                math.clamp(newPos.X, 0, screenSize.X - btnSize.X),
                math.clamp(newPos.Y, 0, screenSize.Y - btnSize.Y)
            )
            
            mobileMenuButton.Position = UDim2.new(0, newPos.X, 0, newPos.Y)
        end
    end)
    
    mobileMenuButton.MouseButton1Click:Connect(function()
        if not wasDragged then
            toggleMenu()
        end
    end)
end

    if gp then return end
    if input.KeyCode==Enum.KeyCode.RightShift or input.KeyCode==Enum.KeyCode.RightControl then toggleMenu()end
    if listeningFor then
        local kb=kbBtns[listeningFor]
        if kb then local bind=Keybinds[listeningFor]
            if bind then bind.key=input.KeyCode;kb.Text=keyName(bind.key);kb.BackgroundColor3=Color3.fromRGB(22,4,35);listeningFor=nil end
        end
        return
    end
    for id,bind in pairs(Keybinds)do
        if input.KeyCode==bind.key and not bind.holdMode then
            if id=="aimbot"then Features.setAimbot(not Features.aimbotEnabled)end
            if id=="silentAim"then Features.setSilentAim(not Features.silentAimEnabled)end
            if id=="esp"then Features.setESP(not Features.espEnabled)end
            if id=="infiniteJump"then Features.setInfiniteJump(not Features.infiniteJumpEnabled)end
            if id=="flick"then doFlick()end
            if id=="unlockAll"then Features.setUnlockAll(not Features.unlockAllEnabled)end
        end
    end
end)

end)
